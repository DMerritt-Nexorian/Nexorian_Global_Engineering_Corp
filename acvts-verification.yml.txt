name: NIST ACVTS Automated Verification Run

on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

jobs:
  acvts-test-run:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Codebase
        uses: actions/checkout@v4

      - name: Setup Python Runtime
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install Dependencies
        run: |
          python -m pip install --upgrade pip
          pip install requests

      - name: Stage mTLS Certificates & Keys
        run: |
          mkdir -p certs
          echo "${{ secrets.ACVTS_CLIENT_CERT }}" > certs/client.crt
          echo "${{ secrets.ACVTS_CLIENT_KEY }}" > certs/client.key
          echo "${{ secrets.ACVTS_TOTP_KEY }}" > certs/totp.key

      - name: Run NIST ACVTS Verification Harness
        run: |
          python scripts/acvts_harness.py
